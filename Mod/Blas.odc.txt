(** Биндинги cuBLAS по cublas_v2.h (CUDA 12), SysV amd64.
	ВНИМАНИЕ: все символы v2 API в .so с суффиксом _v2
	(cublasCreate_v2, cublasSgemm_v2, ...).
	Матрицы column-major (Fortran), float = SHORTREAL. *)

	CONST
		success* = 0;	(* CUBLAS_STATUS_SUCCESS *)

		(* cublasOperation_t *)
		opN* = 0;
		opT* = 1;
		opC* = 2;

	TYPE
		Status* = INTEGER;
		Handle* = LONGINT;	(* cublasHandle_t = указатель *)
		DevicePtr* = LONGINT;

	PROCEDURE [ccall] cublasCreate_v2* (VAR handle: Handle): Status;
	PROCEDURE [ccall] cublasDestroy_v2* (handle: Handle): Status;
	PROCEDURE [ccall] cublasGetVersion_v2* (handle: Handle; VAR version: INTEGER): Status;

	(* C = alpha * op(A) * op(B) + beta * C, single precision, column-major *)
	PROCEDURE [ccall] cublasSgemm_v2* (handle: Handle;
		transa, transb, m, n, k: INTEGER;
		VAR alpha: SHORTREAL; A: DevicePtr; lda: INTEGER;
		B: DevicePtr; ldb: INTEGER;
		VAR beta: SHORTREAL; C: DevicePtr; ldc: INTEGER): Status;

	(* y = alpha * x + y, single precision *)
	PROCEDURE [ccall] cublasSaxpy_v2* (handle: Handle; n: INTEGER;
		VAR alpha: SHORTREAL; x: DevicePtr; incx: INTEGER;
		y: DevicePtr; incy: INTEGER): Status;

	(* dot = x . y, single precision *)
	PROCEDURE [ccall] cublasSdot_v2* (handle: Handle; n: INTEGER;
		x: DevicePtr; incx: INTEGER; y: DevicePtr; incy: INTEGER;
		VAR result: SHORTREAL): Status;

END CudaBlas.
