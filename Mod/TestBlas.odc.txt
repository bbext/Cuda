(** Smoke-тест CudaBlas: sgemm 2x2, saxpy, sdot на GPU.
	Запуск из консоли:  echo 'CudaTestBlas.Go' | ./run-bb64 --console *)

	IMPORT SYSTEM, Console, Strings, CudaRt, CudaBlas;

	VAR bad: INTEGER;

	PROCEDURE WriteInt (x: LONGINT);
		VAR t: ARRAY 40 OF CHAR;
	BEGIN
		Strings.IntToString(x, t); Console.WriteStr(t)
	END WriteInt;

	PROCEDURE CHK (IN s: ARRAY OF CHAR; st: INTEGER);
	BEGIN
		IF st # CudaBlas.success THEN
			INC(bad); Console.WriteStr("BAD "); Console.WriteStr(s);
			Console.WriteStr(" status="); WriteInt(st); Console.WriteLn
		END
	END CHK;

	PROCEDURE CHKR (IN s: ARRAY OF CHAR; got, exp: SHORTREAL);
	BEGIN
		IF ABS(got - exp) > 0.001 THEN
			INC(bad); Console.WriteStr("BAD "); Console.WriteStr(s);
			Console.WriteStr(" got*1000="); WriteInt(ENTIER(got * 1000));
			Console.WriteStr(" exp*1000="); WriteInt(ENTIER(exp * 1000)); Console.WriteLn
		END
	END CHKR;

	(* host -> device: данные dyn-массива = ADR(a[0]) (KB/CudaSubsystem.md) *)
	PROCEDURE Load (IN h: ARRAY OF SHORTREAL): CudaRt.DevicePtr;
		VAR dev: CudaRt.DevicePtr; n: LONGINT;
	BEGIN
		n := LEN(h) * 4;
		CHK("malloc", CudaRt.cudaMalloc(dev, n));
		CHK("memcpy H2D", CudaRt.cudaMemcpy(dev, SYSTEM.ADR(h[0]), n, CudaRt.memcpyHostToDevice));
		RETURN dev
	END Load;

	PROCEDURE Store (dev: CudaRt.DevicePtr; OUT h: ARRAY OF SHORTREAL);
	BEGIN
		CHK("memcpy D2H", CudaRt.cudaMemcpy(SYSTEM.ADR(h[0]), dev, LEN(h) * 4, CudaRt.memcpyDeviceToHost))
	END Store;

	PROCEDURE Go*;
		VAR h: CudaBlas.Handle; ver: INTEGER;
			dA, dB, dC: CudaRt.DevicePtr;
			a, b, c, x, y: ARRAY 4 OF SHORTREAL;
			alpha, beta, dot: SHORTREAL;
	BEGIN
		bad := 0;
		CHK("create", CudaBlas.cublasCreate_v2(h));
		CHK("version", CudaBlas.cublasGetVersion_v2(h, ver));
		Console.WriteStr("cublas version="); WriteInt(ver DIV 1000);
		Console.WriteStr("."); WriteInt((ver MOD 1000) DIV 10); Console.WriteLn;

		(* sgemm: A=[[1,2],[3,4]] B=[[5,6],[7,8]] (column-major!) -> C=[[19,22],[43,50]] *)
		a[0] := 1; a[1] := 3; a[2] := 2; a[3] := 4;
		b[0] := 5; b[1] := 7; b[2] := 6; b[3] := 8;
		dA := Load(a); dB := Load(b); dC := Load(c);
		alpha := 1; beta := 0;
		CHK("sgemm", CudaBlas.cublasSgemm_v2(h, CudaBlas.opN, CudaBlas.opN,
			2, 2, 2, alpha, dA, 2, dB, 2, beta, dC, 2));
		Store(dC, c);
		CHKR("C[0,0]", c[0], 19); CHKR("C[1,0]", c[1], 43);
		CHKR("C[0,1]", c[2], 22); CHKR("C[1,1]", c[3], 50);

		(* saxpy: x=[1,2,3] y=[4,5,6] alpha=2 -> y=[6,9,12] *)
		x[0] := 1; x[1] := 2; x[2] := 3; y[0] := 4; y[1] := 5; y[2] := 6;
		CHK("free A", CudaRt.cudaFree(dA)); CHK("free B", CudaRt.cudaFree(dB));
		dA := Load(x); dB := Load(y); alpha := 2;
		CHK("saxpy", CudaBlas.cublasSaxpy_v2(h, 3, alpha, dA, 1, dB, 1));
		Store(dB, y);
		CHKR("y[0]", y[0], 6); CHKR("y[1]", y[1], 9); CHKR("y[2]", y[2], 12);

		(* sdot: [1,2,3] . [6,9,12] = 6+18+36 = 60 *)
		CHK("sdot", CudaBlas.cublasSdot_v2(h, 3, dA, 1, dB, 1, dot));
		CHKR("dot", dot, 60);

		CHK("free x", CudaRt.cudaFree(dA)); CHK("free y", CudaRt.cudaFree(dB));
		CHK("free C", CudaRt.cudaFree(dC));
		CHK("destroy", CudaBlas.cublasDestroy_v2(h));
		IF bad = 0 THEN Console.WriteStr("CudaTestBlas OK")
		ELSE Console.WriteStr("CudaTestBlas FAILURES") END;
		Console.WriteLn
	END Go;

END CudaTestBlas.
